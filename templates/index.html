function analyze(candles, symbol, tf) { if (!candles) return { tf, status: "DATA_FAILURE", reason: "Fetch returned null", price: 0 }; if (candles.error) return { tf, status: "DATA_FAILURE", reason: candles.error, price: 0 }; if (candles.length < 20) return { tf, status: "DATA_FAILURE", reason: `Insufficient candles (${candles.length}/20)`, price: 0 }; const lastCandle = candles[candles.length - 1]; if (!lastCandle) return { tf, status: "DATA_FAILURE", reason: "Last candle missing", price: 0 }; if (lastCandle.close <= 0 || lastCandle.high <= 0 || lastCandle.low <= 0) { return { tf, status: "DATA_FAILURE", reason: `Invalid price`, price: 0 }; } const highs = candles.map(c => c.high); const lows = candles.map(c => c.low); const resistance = Math.max(...highs); const support = Math.min(...lows); const current = lastCandle.close; const range = resistance - support; if (range <= 0) return { tf, status: "RANGE_FAILURE", reason: "Range width is 0", price: current }; const rangeWidthPct = (range / support) * 100; if (rangeWidthPct < MIN_RANGE_THRESHOLD_PCT) { return { tf, status: "RANGE_FAILURE", reason: `Range too narrow (${rangeWidthPct.toFixed(3)}% < 0.5%)`, price: current }; } const distToRes = ((resistance - current) / current) * 100; const distToSup = ((current - support) / current) * 100; let active_level = "MIDDLE"; if (distToRes < distToSup) active_level = "RESISTANCE"; else if (distToSup < distToRes) active_level = "SUPPORT"; // ------------------ SAFETY INVALIDATION CHECK ------------------ // If the live price has ALREADY broken support or resistance, the setup is DEAD. // We mark it as VALIDATED so it stays in the table, but we force a neutral, safe state. if (current <= support) { return { tf, status: "VALIDATED", reason: "N/A", price: current, resistance, support, distRes: distToRes.toFixed(2), distSup: "0.00", // Already at/below rangeWidthPct: rangeWidthPct.toFixed(3), active_level: "SUPPORT", winner: "SELLERS", setup: "INVALIDATED - BREAKDOWN CONFIRMED", bullishCandles: 0, bearishCandles: 5, // Force bearish buyer_score: "0.0", seller_score: "100.0", margin: "100" }; } if (current >= resistance) { return { tf, status: "VALIDATED", reason: "N/A", price: current, resistance, support, distRes: "0.00", distSup: distToSup.toFixed(2), // Already at/above rangeWidthPct: rangeWidthPct.toFixed(3), active_level: "RESISTANCE", winner: "BUYERS", setup: "INVALIDATED - BREAKOUT CONFIRMED", bullishCandles: 5, bearishCandles: 0, // Force bullish buyer_score: "100.0", seller_score: "0.0", margin: "100" }; } // ------------------------------------------------------------- let buyer_score = 0, seller_score = 0; let bullishCandles = 0, bearishCandles = 0; const lookback = candles.slice(-5); const maxBody = Math.max(...lookback.map(c => Math.abs(c.close - c.open))); lookback.forEach(c => { const body = Math.abs(c.close - c.open); const upperWick = c.high - Math.max(c.close, c.open); const lowerWick = Math.min(c.close, c.open) - c.low; if (c.close > c.open) { bullishCandles++; buyer_score += (body / maxBody) * 10; } else if (c.close < c.open) { bearishCandles++; seller_score += (body / maxBody) * 10; } if (active_level === "RESISTANCE") { if (c.high >= resistance - (range * 0.05)) { if (upperWick > body) seller_score += 3; else if (c.close > resistance) buyer_score += 5; else seller_score += 1; } } else if (active_level === "SUPPORT") { if (c.low <= support + (range * 0.05)) { if (lowerWick > body) buyer_score += 3; else if (c.close < support) seller_score += 5; else buyer_score += 1; } } }); let winner = "NO CLEAR WINNER"; let margin = 0; if (buyer_score > seller_score) { margin = ((buyer_score - seller_score) / ((buyer_score + seller_score) / 2)) * 100; if (margin >= WINNER_MARGIN_PCT) winner = "BUYERS"; } else if (seller_score > buyer_score) { margin = ((seller_score - buyer_score) / ((buyer_score + seller_score) / 2)) * 100; if (margin >= WINNER_MARGIN_PCT) winner = "SELLERS"; } let setup = "NO CLEAR SETUP"; if (active_level === "RESISTANCE" && winner === "BUYERS") setup = "BREAKOUT POTENTIAL"; else if (active_level === "RESISTANCE" && winner === "SELLERS") setup = "REJECTION POTENTIAL"; else if (active_level === "SUPPORT" && winner === "BUYERS") setup = "BOUNCE POTENTIAL"; else if (active_level === "SUPPORT" && winner === "SELLERS") setup = "BREAKDOWN POTENTIAL"; return { tf, status: "VALIDATED", reason: "N/A", price: current, resistance, support, distRes: distToRes.toFixed(2), distSup: distToSup.toFixed(2), rangeWidthPct: rangeWidthPct.toFixed(3), active_level, winner, setup, bullishCandles, bearishCandles, buyer_score: buyer_score.toFixed(1), seller_score: seller_score.toFixed(1), margin: margin.toFixed(0) }; }
